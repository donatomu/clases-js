create table autor(
	Id_autor int auto_increment primary key, 
    Nombre_autor varchar(30)
);

create table editorial(
	Cod_Ed int auto_increment primary key,
    Nombre_Ed varchar(25),
    Pais_Ed varchar(30),
    Telefono_Ed varchar(40)
);
 
 create table lector(
	Num_Tarjeta int auto_increment primary key,
    Nombre_Lector varchar (20),
    direccion_Lector varchar(30),
    Telefono_Lector varchar(20)
 );
 
 create table libro(
	Id_libro int auto_increment primary key,
    titulo varchar(20),
    Cod_Ed int,
    FOREIGN KEY (Cod_Ed) REFERENCES editorial(Cod_Ed)
 );
 
 create table autor_libro(
	Id_libro int,
    Id_autor int,
    FOREIGN KEY (Id_libro) REFERENCES libro(Id_libro),
     FOREIGN KEY (Id_autor) REFERENCES autor(Id_autor)
 );
 
 create table prestamos(
	Id_prestamo int auto_increment primary key,
    Num_Tarjeta int,
    fecha_salida date,
    FOREIGN KEY (Num_Tarjeta) REFERENCES lector(Num_Tarjeta)
 );
 create table detalle_prestamo(
	Id_prestamo int,
    Id_libro int,
    fecha_devol date,
    mora int,
    devuelto boolean,
    FOREIGN KEY (Id_prestamo) REFERENCES prestamos(Id_prestamo),
	FOREIGN KEY (Id_libro) REFERENCES libro(Id_libro)
 );
